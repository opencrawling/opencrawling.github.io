Official Client SDKs · Java 25 & Python 3.11+

Programmatic Ingestion Control for Java & Python

Strongly typed official SDKs — oc-java-client-sdk (Java 25 & Spring Boot) and opencrawling-sdk (Python 3.11+ sync & async over httpx and Pydantic v2) — delivering 100% Platform API v1 (/api/v1) parity, OIS Direct Ingestion, optional gRPC Push streaming, RFC 8693 AI agent identity delegation, and Lakehouse analytics.

Explore Code Examples Install via pip or Maven Download GA & RC Artifacts

100% Platform API v1 Parity

Every one of the 65 /api/v1 operations is covered in both Java (PlatformClient) and Python (OpenCrawlingClient & AsyncOpenCrawlingClient), verified continuously in CI against OpenAPI 3.1.

Built-In REST Resilience & ETags

Versioned<T> ETag concurrency (If-Match / If-None-Match), automatic UUIDv4 Idempotency-Key on POST, jittered RetryPolicy honoring Retry-After, lossless settings round-trips, and structured RFC 9457 problem errors.

Enterprise OAuth 2.0 & RFC 8693 Agents

First-class ClientCredentialsTokenProvider, RefreshTokenProvider, DeviceCodeLogin (RFC 8628), and RFC 8693 TokenExchangeProvider so AI agents act on behalf of end users with verified act.sub provenance and zero secret leakage.

OIS Direct Ingest & gRPC Push

Push typed OisDocument UPSERT and DELETE tombstones with fail-closed SecurityConfig ACL rules via HTTP (POST /api/v1/ingest/ois/{jobId}) or high-throughput gRPC streaming (opencrawling-sdk[grpc]).

Lakehouse Analytics & DataFrames

Stream Parquet, CSV, Avro, and JSON execution audit reports directly to disk with constant memory usage, run read-only DuckDB SQL queries, and convert results into pyarrow, pandas, or polars DataFrames.

Spring Boot & Asyncio Ready

Zero-dependency Java 25 HttpClient with Spring Boot auto-configuration (opencrawling.client.*), alongside PEP 561 strictly typed Python sync/async clients built for FastAPI, LangGraph, LlamaIndex, and Airflow.

Interactive Usage Examples

platform_v1_sync.py 
from opencrawling import OpenCrawlingClient, RequestOptions

with OpenCrawlingClient(
    base_url="http://localhost:8080",
    client_credentials={
        "token_endpoint": "https://idp.example.com/realms/opencrawling/protocol/openid-connect/token",
        "client_id": "opencrawling-automation",
        "client_secret": "...",
    },
) as client:
    # 1. Read a job along with its strong ETag
    versioned_job = client.jobs.get("1")
    print(f"Job '{versioned_job.data.name}' (ETag: {versioned_job.etag})")

    # 2. Patch conditionally with If-Match (RFC 7396 Merge Patch)
    client.jobs.patch(
        "1",
        {"path": "/Sites/finance/documentLibrary"},
        options=RequestOptions(if_match=versioned_job.etag),
    )

    # 3. Start a job run (automatic Idempotency-Key) and wait for completion
    started = client.jobs.start("1")
    completed = client.job_runs.wait_for_completion(started.data.execution_id, timeout=300.0)

    # 4. Stream Parquet report to disk & query the Lakehouse with DuckDB
    client.executions.export_to(completed.execution_id, "parquet", "./audit.parquet")
    summary = client.executions.query(
        "SELECT status, COUNT(*) AS docs FROM read_parquet('mode=RAG/**/*.parquet') GROUP BY status"
    )
    print(summary.to_dicts())
async_agent_rfc8693.py 
import asyncio
from opencrawling import AsyncOpenCrawlingClient, DiscoveryDryRunRequest, DocumentIngestRequest
from opencrawling.auth import TokenExchangeProvider

async def agent_workflow(user_subject_token: str) -> None:
    # RFC 8693 OAuth 2.0 Token Exchange: agent acts on behalf of end user (act.sub)
    delegated_auth = TokenExchangeProvider(
        token_endpoint="https://idp.example.com/realms/opencrawling/protocol/openid-connect/token",
        client_id="research-agent",
        client_secret="...",
        subject_token=user_subject_token,
    )

    async with AsyncOpenCrawlingClient("http://localhost:8080", token_provider=delegated_auth) as client:
        # 1. Preview matching documents under the user's source ACLs
        preview = await client.discovery.dry_run(
            "SharePoint_Prod",
            DiscoveryDryRunRequest(scope="/Drives/Legal", sample_size=25),
        )
        print(f"Estimated readable docs: {preview.match_count} ({preview.count_accuracy})")

        # 2. Dynamically ingest a single document on demand
        accepted = await client.dynamic_jobs.ingest_document(
            DocumentIngestRequest.upsert(
                connector="SharePoint_Prod",
                document="01BYE5RZ6QN3ZWBTUFOFD3GSPGOHDJD36K",
            )
        )
        print(f"Dynamic job triggered: {accepted.job_id} ({accepted.state})")

        # 3. Iterate authorized OIS metadata sidecars lazily
        async for sidecar in client.sidecars.iter("S3_Migration_Sink", prefix="legal/2026/"):
            print(sidecar.get("id"))

asyncio.run(agent_workflow("eyJ..."))
ois_and_grpc_push.py 
from opencrawling import OpenCrawlingClient
from opencrawling.ingest import (
    GrpcPushClient,
    OisContent,
    OisDocument,
    PermissionRule,
    SecurityConfig,
)

# 1. Direct HTTP OIS push (POST /api/v1/ingest/ois/{jobId}) with fail-closed ACLs
with OpenCrawlingClient("http://localhost:8080", bearer_token="ois-ingest-secret") as client:
    doc = OisDocument.upsert(
        id="custom://crm/accounts/ACME-2026-Q4",
        content=OisContent.markdown("# ACME Corp Q4 Renewal\nAnnual value: $480,000."),
        metadata={"account_id": "ACME-001", "region": "EMEA"},
        security=SecurityConfig.fail_closed([
            PermissionRule.allow_group("sales-emea"),
            PermissionRule.deny_user("contractor@example.com"),
        ]),
    )
    receipt = client.ingest.push_ois("crm-rag-pipeline", doc)
    print(f"Accepted={receipt.is_accepted} ({receipt.status})")

# 2. High-throughput gRPC Push streaming (pip install "opencrawling-sdk[grpc]")
with GrpcPushClient("localhost:9096", bearer_token="grpc-push-secret") as grpc_client:
    resp = grpc_client.push_document("crm-rag-pipeline", doc)
    print(f"gRPC status: {resp.status.name}")
PlatformApiV1Example.java 
import org.opencrawling.sdk.OpenCrawlingClient;
import org.opencrawling.sdk.v1.*;
import org.opencrawling.sdk.v1.model.*;
import java.time.Duration;
import java.util.Map;

OpenCrawlingClient client = OpenCrawlingClient.builder()
        .baseUrl("http://localhost:8080")
        .clientCredentials(
                "https://idp.example.com/realms/opencrawling/protocol/openid-connect/token",
                "opencrawling-automation",
                System.getenv("OPENCRAWLING_CLIENT_SECRET"),
                null)
        .build();

PlatformClient api = client.platform();

// Idempotent creation & ETag-protected JSON Merge Patch
Versioned<Job> job = api.jobs().create(
        JobInput.of("HR contracts", "FileSystem_Local", "PGVector_Output", "/data/hr"),
        RequestOptions.idempotent("hr-contracts-1"));
api.jobs().patch(job.value().id(), Map.of("path", "/data/hr/2026"), RequestOptions.ifMatch(job.etag()));

// Start run and wait for completion
Versioned<JobRun> run = api.jobs().start(job.value().id(), RequestOptions.idempotent("hr-run-1"));
JobRunHistory done = api.jobRuns().waitForCompletion(run.value().executionId(), Duration.ofSeconds(2), Duration.ofMinutes(5));
System.out.println("Completed state: " + done.state());
OisDirectIngestExample.java 
import org.opencrawling.sdk.ingest.*;
import java.util.List;
import java.util.Map;

OisIngestClient ingest = client.ingest().withToken(System.getenv("OPENCRAWLING_OIS_TOKEN"));

OisIngestResponse upserted = ingest.push("1", OisDocument.upsert(
        "doc-2026-001",
        OisContent.markdown("# Q1 Financial Summary\nQuarterly revenue reached $42M."),
        Map.of("department", "finance"),
        SecurityConfig.failClosed(List.of(
                PermissionRule.allowGroup("finance-analysts"),
                PermissionRule.denyGroup("contractors"))),
        OisSource.of("custom-erp", "erp-prod-eu")));

System.out.println("Accepted: " + upserted.isAccepted());

      token-uri: https://idp.example.com/realms/opencrawling/protocol/openid-connect/token
      client-id: opencrawling-automation
      client-secret: ${OPENCRAWLING_CLIENT_SECRET}

// Spring Service
import org.opencrawling.sdk.OpenCrawlingClient;
import org.opencrawling.sdk.v1.RequestOptions;
import org.springframework.stereotype.Service;

@Service
public class DocumentIngestionService {

    private final OpenCrawlingClient client;

    public DocumentIngestionService(OpenCrawlingClient client) {
        this.client = client;
    }

    public void triggerJob(String jobId) {
        client.platform().jobs().start(jobId, RequestOptions.none());
    }
}
